## Usage

```tsx
import { Field, FieldDescription, FieldLabel } from "@/components/ui/field";
```

```tsx
<Field>
  <FieldLabel htmlFor="name">Name</FieldLabel>
  <Input id="name" />
  <FieldDescription>Your full name.</FieldDescription>
</Field>
```
